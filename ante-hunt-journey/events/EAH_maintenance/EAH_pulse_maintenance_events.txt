# When the inquisition arrives, hunting those dastardly antediluvians is off the table

# I don't know the convention for numbering this sort of thing so I chose 1
# Currently this event just checks if it's endgame time then calls the clean-up event
EAH_maintenance.1 = {
	hidden = yes
	scope = none

	if = {
		limit = {
			current_year = inquisition_spawn_year
		}
		trigger_event = {
			id = EAH_endgame.1
			days = { 0 }
		}
	}
}
