namespace = EAH_debug

# Debug: Mark all Antediluvians as dead
EAH_debug.1 = {
	type = character_event
	title = "DEBUG: Kill All Antediluvians"
	desc = "This will mark all Antediluvians as dead for testing purposes."
	
	immediate = {
		set_global_variable = {
			name = variable_POD_absi_dead
			value = yes
		}
		set_global_variable = {
			name = variable_POD_lasombra_dead
			value = yes
		}
		set_global_variable = {
			name = variable_POD_set_dead
			value = yes
		}
		set_global_variable = {
			name = variable_POD_tremere_dead
			value = yes
		}
		set_global_variable = {
			name = variable_POD_troile_dead
			value = yes
		}
		set_global_variable = {
			name = variable_POD_tzimisce_dead
			value = yes
		}
	}
	
	option = {
		name = "All Antediluvians are now dead"
		
		show_as_tooltip = {
			set_global_variable = {
				name = variable_POD_absi_dead
				value = yes
			}

				name = variable_POD_lasombra_dead
				value = yes
			}

				name = variable_POD_set_dead
				value = yes
			}

				name = variable_POD_tremere_dead
				value = yes
			}
			set_global_variable = {
				name = variable_POD_troile_dead
				value = yes
			}
			set_global_variable = {
				name = variable_POD_tzimisce_dead
				value = yes
			}
		}
	}
}
