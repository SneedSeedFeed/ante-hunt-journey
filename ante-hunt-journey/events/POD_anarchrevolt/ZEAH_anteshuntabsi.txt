#Meeting Baba Yaga
POD_anteshuntabsi.5 = {
	type = character_event
	content_source = dlc_POD
	title = POD_anteshuntabsi.5.t
	desc = POD_anteshuntabsi.5.desc
	theme = nosferatu_coa
	
	option = {
		name = POD_anteshuntabsi.5.a
	
		if = {
			limit = {
				is_unbondable_trigger = yes
			}
			add_character_modifier = {
				modifier = babayaga_bound_modifier
				years = 10
			}
		}
		
		else = {
			add_character_modifier = {
				modifier = babayaga_bound_modifier
				years = 50
			}
		}
		
		trigger_event = {
			id = POD_anteshuntabsi.6
			days = { 10 15 }
		}
	}
	
	option = {
		name = POD_anteshuntabsi.5.b

		eah_end_ante_journey_effect = { TYPE = eah_hunt_ante_absi }

		duel = {
			skill = prowess
			value = 70
			20 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = 0.5
				}
				desc = POD_anteshuntabsi.5.effect.success.desc
			}
			20 = {
				compare_modifier = {
					value = scope:duel_value
					multiplier = -0.5
				}
				desc = POD_anteshuntabsi.5.critical.failure.effect.desc

				send_interface_toast = {
					title = POD_anteshuntabsi.5.critical.failure
					left_icon = root

					death = {
						death_reason = death_execution
					}
				}
			}
		}
		custom_tooltip = anteabsi_end_baba
	}			
}

POD_anteshuntabsi.20 = {
	type = character_event
	content_source = dlc_POD
	title = POD_anteshuntabsi.20.t
	desc = POD_anteshuntabsi.20.desc
	theme = nosferatu_coa

	option = {
		name = POD_anteshuntabsi.20.a
			
		POD_remove_vampire_generation_traits_effect = yes	
		add_trait_force_tooltip = vampgeneration3
		
		assign_animalism_effect = yes
		assign_obfuscate_effect = yes
		assign_potence_effect = yes
		
		POD_give_diablerist_secret_or_trait_effect = yes

		remove_character_modifier = babayaga_bound_modifier
	
		if = {
			limit = {
				character:505075 = { is_alive = yes }
			}
		character:505075 = {
		death = { 
			death_reason = death_murder
			killer = root
		}
		}
		}
			
		add_prestige = massive_prestige_value
		add_prestige_level = 2				
		
		set_global_variable = {
			name = variable_POD_absi_dead
			value = yes
		}

		eah_end_ante_journey_effect = { TYPE = eah_hunt_ante_absi }
		
		if = {
			limit = { has_character_modifier = pseudo_third_modifier }
			remove_character_modifier = pseudo_third_modifier
		}
		
		if = {
			limit = { NOT = { has_character_modifier = bloodpotency10_modifier } }
			POD_remove_bloodpotency_modifiers_effect = yes
			add_character_modifier = bloodpotency10_modifier
		}
		
		add_character_modifier = {
			modifier = slayer_of_gods
			years = 50
			}	

		add_character_modifier = {
			modifier = young_third_modifier
			}	
			
		character:505011 = {
			set_death_reason = {
				death_reason = death_diablerie
				killer = root
			}
		}
		POD_achiev_nomnom_check = yes	
	}
}
