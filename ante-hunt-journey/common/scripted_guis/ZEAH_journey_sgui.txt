eah_hunt_ante_requirements = {
	scope = character
	is_shown = { 
		AND = { 
			NOT = { global_var:variable_EAH_hunt_closed ?= yes }
			POD_is_vampire_trigger = yes 
			NOT = { has_character_modifier = young_third_modifier }
		} 
	}
	is_valid = { eah_not_on_ante_hunt_trigger = yes }
}
eah_hunt_ante_on_end = { scope = character effect = { } }
eah_hunt_ante_travel_phase = { scope = character effect = { } }
